def set_cell_margins(cell, top=50, bottom=50, left=70, right=70):
    tcPr = cell._tc.get_or_add_tcPr()
    tcMar = OxmlElement('w:tcMar')
    for m, val in [('w:top', top), ('w:bottom', bottom), ('w:left', left), ('w:right', right)]:
        node = OxmlElement(m)
        node.set(qn('w:w'), str(val))
        node.set(qn('w:type'), 'dxa')
        tcMar.append(node)
    tcPr.append(tcMar)

def set_cell_shading(cell, color_hex):
    tcPr = cell._tc.get_or_add_tcPr()
    for s in tcPr.xpath('w:shd'):
        tcPr.remove(s)
    shd = parse_xml(f'<w:shd {nsdecls("w")} w:fill="{color_hex}"/>')
    tcPr.append(shd)

def set_cell_vcenter(cell):
    tcPr = cell._tc.get_or_add_tcPr()
    for v in tcPr.xpath('w:vAlign'):
        tcPr.remove(v)
    vAlign = OxmlElement('w:vAlign')
    vAlign.set(qn('w:val'), 'center')
    tcPr.append(vAlign)

def set_row_cant_split(row):
    trPr = row._tr.get_or_add_trPr()
    cantSplit = OxmlElement('w:cantSplit')
    trPr.append(cantSplit)

def style_run(run, font_name='Arial', size_pt=10.0, bold=False, italic=False, color_rgb=(0x3B, 0x38, 0x38)):
    run.font.name = font_name
    run.font.size = Pt(size_pt)
    run.bold = bold
    run.italic = italic
    if color_rgb:
        run.font.color.rgb = RGBColor(*color_rgb)

def generate_document_by_language(template_path, language_mode, output_path):
    # language_mode: "id", "en", atau "bi"
    doc = docx.Document(template_path)
    if language_mode == "id":
        build_indonesian_content(doc)
    elif language_mode == "en":

    elif language_mode == "bi":
        build_bilingual_content(doc)
